import matplotlib.pyplot as plt

battery = [1000, 2000, 3000, 4000, 5000]
backup = [4, 8, 12, 16, 20]

def regression(x, y):
    n = len(x)
    b1 = (n*sum(x[i]*y[i] for i in range(n))-sum(x)*sum(y)) / (n*sum(i*i for i in x)-sum(x)**2)
    b0 = (sum(y)-b1*sum(x))/n
    return b0, b1

b0, b1 = regression(battery, backup)

new = 4500
pred = b0 + b1 * new

print("Predicted Backup:", round(pred, 2), "Hours")

plt.scatter(battery, backup, label="Actual Data")
plt.plot(battery, [b0+b1*i for i in battery], label="Regression Line")
plt.scatter(new, pred, marker="*", s=180, label="Prediction")
plt.text(new, pred, f"{pred:.2f} Hours")
plt.title("Battery Backup Prediction")
plt.xlabel("Battery")
plt.ylabel("Backup Hours")
plt.legend()
plt.grid()

plt.savefig("battery_backup_chart.png")
plt.show()
