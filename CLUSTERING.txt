import numpy as np 
from sklearn.cluster import KMeans
data = np.array([
    [10, 15, 2, 1],
    [12, 17, 3, 1],
    [15, 20, 4, 2],
    [70, 75, 20, 3],
    [72, 78, 21, 3],
    [75, 80, 22, 4],
    [40, 45, 10, 2],
    [43, 48, 11, 2]
])
model = KMeans (n_clusters = 3, random_state = 42, n_init = 10)
model.fit(data)
def distance(X, c):
    return np.sqrt(np.sum((X - c)**2))
for i, X in enumerate (data):
    d = [distance(X, c)for c in model.cluster_centers_]
    print("crime", i+1, "->cluster", np.argmin(d)+1)
