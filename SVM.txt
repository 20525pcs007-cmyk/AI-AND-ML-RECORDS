import pandas as pd
from sklearn.svm import SVR

df = pd.DataFrame({
    "Distance":[2,5,8,10,3,7,12,4,9,6],
    "Traffic":[1,2,3,3,1,2,3,1,3,2],
    "Weather":[1,2,3,4,1,2,5,2,4,3],
    "Items":[1,2,3,4,1,2,5,2,4,3],
    "Time":[15,30,45,60,20,35,75,25,55,40]
})

X = df[["Distance","Traffic","Weather","Items"]]
y = df["Time"]

model = SVR(kernel="linear")
model.fit(X, y)
W = model.coef_[0]
b = model.intercept_[0]
distance, traffic, weather, items = 6, 3, 2, 3
prediction = (W[0]*distance+W[1]*traffic+W[2]*weather+W[3]*items+b)
print("Weights:", W)
print("Bias:", b)
print("Predicted Delivery Time:", round(prediction, 2), "minutes")
