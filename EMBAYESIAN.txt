import numpy as np 
data = np.array([[20, 2, 10],[40, 5, 20],[60, 8, 30],[80, 12,40]])
p = 0.5
m_block = np.mean(data[2:], axis=0)
m_clear = np.mean(data[:2], axis=0)
for i in range(5):
    db = np.sum((data-m_block)**2, axis=1)
    dc = np.sum((data-m_clear)**2, axis=1)
    block = p*np.exp(-db/2)
    clear = (1-p)*np.exp(-dc/2)
    r = block/(block+clear)
    p = np.mean(r)
    m_block = np.sum(r[:, None]*data, axis=0)/sum(r)
    m_clear = np.sum((1-r)[:, None]*data, axis=0)/sum(1-r)
    rain, waste, water = 75, 11, 38
    X = np.array([rain, waste, water])
    block = p*np.exp(-np.sum((X-m_block)**2)/2)
    clear = (1-p)*np.exp(-np.sum((X-m_clear)**2)/2)
    risk = block/(block+clear)
    print("Rainfall:", rain, "mm")
    print("Waste:", waste, "kg")
    print("Water Level:", water, "cm")
    print("Blockage Risk:", round(risk, 2))
    print("Result:", "HIGH RISK" if risk >=0.5 else "LOW RISK")
