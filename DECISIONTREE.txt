import pandas as pd
import matplotlib.pyplot as plt
from sklearn.tree import DecisionTreeClassifier
from sklearn.feature_extraction.text import CountVectorizer

df = pd.DataFrame({
    "email":["Win Money","Won Prize","Free Gift","Meeting","Submit Work","Send Report"],
    "spam":[1,1,1,0,0,0]
})

v = CountVectorizer()
X = v.fit_transform(df.email)

model = DecisionTreeClassifier().fit(X, df.spam)

email = ["Won Free Money"]
r = model.predict(v.transform(email))[0]

print("Email:", email[0])
print("Result:", "SPAM" if r else "NOT SPAM")

plt.scatter(df.index, df.spam, s=(df.spam+1)*700, alpha=0.6)
plt.yticks([0,1], ["Not Spam","Spam"])
plt.xlabel("Email Number")
plt.ylabel("Email Type")
plt.title("Email Spam Detection - Bubble Chart")
plt.grid()

plt.savefig("spam_chart.png")
plt.show()
